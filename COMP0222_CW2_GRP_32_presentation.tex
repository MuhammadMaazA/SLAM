\begin{frame}[shrink=5]{Q2(a) Capture and Calibration}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{\color{indigo}Rig and protocol}
      \begin{itemize}\small
        \item Intel RealSense D455, RGB at 848$\times$480, 30\,fps
        \item Auto-exposure locked after 2\,s warm-up; walk 0.3--0.5\,m/s
        \item \textbf{Basement\_1} (indoor): LGF Lift D room, 869 ORB-SLAM2 poses, closed loop with concrete walls on all four sides
        \item \textbf{Outdoor\_1} (outdoor): rear courtyard, brick facade + trees, 983 poses
      \end{itemize}
      \vspace{0.3em}
      \textbf{\color{indigo}Calibration via COLMAP self-calibration}
      \begin{itemize}\small
        \item OPENCV 8-param $(f_x,f_y,c_x,c_y,k_1,k_2,p_1,p_2)$ per sequence
        \item Basement\_1 scratch SIMPLE\_PINHOLE (no prior) confirms intrinsics recoverable from images
        \item Calibrated YAML fed to ORB-SLAM2 (\texttt{ba\_refine\_focal\_length=0} fixes $f$ in later runs)
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      {\footnotesize
      \begin{tabular}{@{}lccr@{}}
        \toprule
        Run & $f_x$ (px) & $f_y$ (px) & $\Delta f_x$\\
        \midrule
        Factory D455       & 426.7 & 426.1 & ---\\
        B1 scratch PINHOLE & 415.4 & 415.4 & $-2.6\,\%$\\
        B1 OPENCV          & 430.6 & 445.8 & $+0.9\,\%$\\
        Outdoor\_1 OPENCV  & 427.9 & 437.2 & $+0.3\,\%$\\
        \bottomrule
      \end{tabular}}
      \vspace{0.4em}
      \begin{itemize}\small
        \item Both OPENCV runs agree with factory $f_x$ within 1\,\%
        \item $f_y$ inflated 2--5\,\%: BA trades $f_y$ against radial coefficients when depth variation is limited
        \item SIMPLE\_PINHOLE $f{=}415.4$\,px is 2.6\,\% below factory (forced $f_x{=}f_y$ on mildly asymmetric sensor)
      \end{itemize}
      \vspace{0.2em}
      \begin{kfbox}
        Self-calibration adequate for low-distortion D455, but inflated $f_y$ causes Outdoor\_1 heading artefact
      \end{kfbox}
    \end{column}
  \end{columns}

        \bottomrule

      \vspace{0.3em}
      {\footnotesize\textbf{Sequences (RPLidar A2M12):}}
      \begin{itemize}\footnotesize
        \item \textbf{Basement\_1}: 7$\times$3\,m room, 1939 scans, best ICP health
        \item \textbf{Floor7\_Hallway}: 7th-floor hall ($\sim$20\,m sides), 1304 scans
        \item \textbf{Outdoor\_1}: ground-floor exit, 1462 scans, ICP freezes at scan~400
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{itemize}\small
        \item \textbf{Only Basement\_1} produces a coherent map at baseline; other two stay degenerate under every variant --- sweep is diagnostic rather than absolute
        \item \textbf{Max-range 2\,m}: cannot reach far walls of 7$\times$3\,m room; translation drops out (rank-deficient normal equations). Outdoors: truncates noisy far returns, improves ICP
        \item \textbf{Angular subsampling}: non-monotone in $n$; rotation is the under-constrained mode in the hill-climber
        \item \textbf{Voxel 0.05\,m}: normalises point density in open hall, but destabilises sparse outdoor scans
        \item \textbf{Skip 33\,\%}: collapses trajectory on B1 (per-scan displacement exceeds 30\,mm search window)
      \end{itemize}
    \end{column}
  \end{columns}

% Q3(b) — Key parameter findings
% ---------------------------------------------------------------------------
\begin{frame}[shrink=5]{Q3(b) Key Parameter Findings}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \includegraphics[width=\linewidth]{plots/q3b_grids_basement_1_max_range.png}\\[3pt]
      \includegraphics[width=\linewidth]{plots/q3b_grids_basement_1_angular_resolution.png}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{itemize}\small
        \item \textbf{Range 2\,m is environment-dependent.} Indoors cannot reach far walls; translation drops out. Outdoors truncates noisy distant returns (glass, vegetation) and ICP improves
        \item \textbf{Angular subsampling is uniformly harmful.} Effect is non-monotone in $n$. Rotation is the under-constrained mode in the hill-climber
        \item \textbf{Voxel downsampling helps in the open hall} by normalising point density, but destabilises sparse outdoor scans (worst result at 0.10\,m)
        \item \textbf{Skip=3 collapses trajectory.} Indoor path length 35\,m $\to$ short; low closure is a truncation artefact. Per-scan displacement exceeds 30\,mm $\times$ 10-iter search window
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[shrink=15]{Q3(c) Loop Closure Detection}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \includegraphics[width=\linewidth]{plots/q3c_basement_1.png}
    \end{column}
    \begin{column}{0.43\textwidth}
      \textbf{Detector}
      \begin{itemize}\scriptsize
        \item Spatial proximity with non-maximum suppression
        \item Arc gate $\ge\min(8\,\text{m},\,0.4{\cdot}\text{arc})$; pose dist $\le 2.0$\,m
        \item ICP overlap score $\ge 0.70$: 70\,\% of scan points within 0.4\,m of reference
      \end{itemize}
      \vspace{0.25em}
      \textbf{Results}
      \begin{itemize}\scriptsize
        \item \textbf{Basement\_1}: 10 closures / 2933 candidates; clean approach-and-drop score pattern
        \item \textbf{Floor7\_Hallway}: 8 closures / 1991 candidates
        \item \textbf{Outdoor\_1}: 0 closures; trajectory frozen, arc gate rejects almost all pairs
      \end{itemize}
      \vspace{0.2em}
      \begin{kfbox}
        Score gap 0.45--0.60 (false) vs 0.75--0.95 (true); threshold 0.70 keeps every genuine revisit across both indoor sequences
      \end{kfbox}
    \end{column}
  \end{columns}
\end{frame}
